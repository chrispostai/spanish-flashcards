    <script>

        // Progress storage and logic

        let progress = JSON.parse(localStorage.getItem('spanish_srs_progress')) || {};

        let studyMode = localStorage.getItem('spanish_study_mode') || 'es-en';

        let currentCard = null;

        let isFlipped = false;


        function setMode(mode) {

            studyMode = mode;

            localStorage.setItem('spanish_study_mode', mode);

            document.querySelectorAll('.mode-btn').forEach(b => b.classList.remove('active'));

            document.getElementById(`mode-${mode}`).classList.add('active');

            loadNextCard();

        }


        function getCardData(word) {

            if (!progress[word.id]) {

                progress[word.id] = { interval: 1, easeFactor: 2.5, dueDate: Date.now(), reviews: 0 };

            }

            return progress[word.id];

        }


        function getDueCards() {

            const now = Date.now();

            return initialWords.filter(w => getCardData(w).dueDate <= now);

        }


        function saveProgress() {

            localStorage.setItem('spanish_srs_progress', JSON.stringify(progress));

        }


        function updateStats() {

            const dueCount = getDueCards().length;

            const studiedCount = Object.keys(progress).filter(id => progress[id].reviews > 0).length;

            document.getElementById('stats').innerText = `Due: ${dueCount} | Studied: ${studiedCount} / ${initialWords.length}`;

        }


        function loadNextCard() {

            const due = getDueCards();

            updateStats();

            if (due.length === 0) {

                document.getElementById('front-label').innerText = "SUCCESS";

                document.getElementById('front-word').innerText = "All caught up! 🎉";

                document.getElementById('back-label').innerText = "SEE YOU TOMORROW";

                document.getElementById('back-word').innerText = "Check back later!";

                document.getElementById('controls').classList.add('hidden');

                currentCard = null;

                return;

            }


            document.getElementById('controls').classList.remove('hidden');

            currentCard = due[0];


            let dir = studyMode;

            if (studyMode === 'mix') dir = Math.random() > 0.5 ? 'es-en' : 'en-es';


            if (dir === 'es-en') {

                document.getElementById('front-label').innerText = "SPANISH";

                document.getElementById('front-word').innerText = currentCard.es;

                document.getElementById('back-label').innerText = "ENGLISH";

                document.getElementById('back-word').innerText = currentCard.en;

            } else {

                document.getElementById('front-label').innerText = "ENGLISH";

                document.getElementById('front-word').innerText = currentCard.en;

                document.getElementById('back-label').innerText = "SPANISH";

                document.getElementById('back-word').innerText = currentCard.es;

            }


            isFlipped = false;

            document.getElementById('card').classList.remove('flipped');

        }


        function flipCard() {

            if (!currentCard) return;

            isFlipped = !isFlipped;

            document.getElementById('card').classList.toggle('flipped', isFlipped);

        }


        function rateCard(rating) {

            if (!currentCard) return;

            const p = getCardData(currentCard);

            const dayMs = 24 * 60 * 60 * 1000;

            if (rating === 'hard') {

                p.interval = 1;

                p.easeFactor = Math.max(1.3, p.easeFactor - 0.2);

            } else if (rating === 'good') {

                p.interval = Math.round(p.interval * p.easeFactor);

            } else if (rating === 'easy') {

                p.easeFactor += 0.15;

                p.interval = Math.round(p.interval * p.easeFactor * 1.3);

            }

            p.dueDate = Date.now() + (p.interval * dayMs);

            p.reviews += 1;

            saveProgress();

            loadNextCard();

        }


        document.querySelectorAll('.mode-btn').forEach(b => b.classList.remove('active'));

        document.getElementById(`mode-${studyMode}`).classList.add('active');

        loadNextCard();

    </script>
